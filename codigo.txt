let prevScrollTop = window.pageYOffset || document.documentElement.scrollTop;
let prevScrollDirection = '';

window.addEventListener('scroll', function() {
  const st = window.pageYOffset || document.documentElement.scrollTop;
  if (st > prevScrollTop && prevScrollDirection !== 'down') {
    console.log('down');
    prevScrollDirection = 'down';
  }
  else if (st < prevScrollTop && prevScrollDirection !== 'up') {
    console.log('up');
    prevScrollDirection = 'up';
  }
  prevScrollTop = st <= 0 ? 0 : st; // for Mobile or negative scrolling
}, false);

let isAnimating = false;

window.addEventListener('wheel', (e) => {
  // detectar scroll horizontal (trackpad) OU scroll vertical convertido
  let delta = e.deltaX !== 0 ? e.deltaX : e.deltaY;

  if (delta > 0 && !isAnimating) {
    // scroll para a direita
    isAnimating = true;

    archive_container.style.transform = 'translateX(0)';
    archive_container.style.transition = 'transform 0.8s ease';

    // bloquear scroll real
    e.preventDefault();

    setTimeout(() => {
      isAnimating = false;
    }, 800);
  }

}, { passive: false });

// data (agora TopoJSON)
let geoMap = "./data/concelhos_continente.json";

window.onload = function () {
    let promises = [
        d3.json(geoMap)
    ];

    Promise.all(promises).then(draw_map);
}

function draw_map(data) {

    let geojson = topojson.feature(
        data[0],
        data[0].objects.ContinenteConcelhos
    );

    let svg = d3.select('.atlas_grid_container')
        .append('svg')
        .attr('class', 'mapa_svg');

    requestAnimationFrame(() => {

        let bounds = svg.node().getBoundingClientRect();

        let width = bounds.width;
        let height = bounds.height;

        svg.attr("viewBox", `0 0 ${width} ${height}`);

        let projection = d3.geoMercator()
            .fitSize([width, height], geojson);

        let path = d3.geoPath()
            .projection(projection);

        svg.append("g")
            .selectAll("path")
            .data(geojson.features)
            .enter()
            .append("path")
            .attr("d", path)
            .attr("fill", "#f5f5f5")
            .attr("stroke", "#333")
            .attr("stroke-width", 0.5);
    });
}

#open_about {
    position: absolute;
    top: 45vh;
    left: 2vw;
    cursor: pointer;
    transform-origin: left center;
}

#open_about.fixed {
    position: fixed;
    top: 5vh;
    z-index: 1000;
}const openAbout = document.getElementById("open_about");
const trigger = document.querySelector(".open_trigger");

let isFixed = false;

window.addEventListener("scroll", () => {

    const triggerTop = trigger.getBoundingClientRect().top;

    // começa a animar 200px antes do trigger
    const start = 200;

    let progress = (start - triggerTop) / start;
    progress = Math.max(0, Math.min(progress, 1));

    const eased = progress * progress;

    const angle = -90 + 90 * eased;
    const moveX = 100 * eased;

    if (progress < 1) {

        // Se estava fixo, volta ao estado inicial
        if (isFixed) {
            openAbout.classList.remove("fixed");
            openAbout.style.left = "";
            isFixed = false;
        }

        openAbout.style.transform =
            `translateX(${moveX}px) rotate(${angle}deg)`;

    } else {

        if (!isFixed) {

            // Guarda a posição EXACTA onde terminou a animação
            const rect = openAbout.getBoundingClientRect();

            openAbout.classList.add("fixed");

            // Mantém exatamente a mesma posição horizontal
            openAbout.style.left = `${rect.left}px`;

            // Estado final
            openAbout.style.transform = "rotate(0deg)";

            isFixed = true;
        }
    }

});
